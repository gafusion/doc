\section{Transport problem}

Steady state, for each energy channel $\sigma \in \{e,i\}$:
%
\begin{equation}
\frac{1}{\vpr(r)} \frac{\partial}{\partial r}
 \left[ \vpr(r) \, Q_\sigma(r) \right] = S_\sigma(r) \; .
\label{eq.transport}
\end{equation}
%
Integrating gives the target flux
%
\begin{equation}
Q^T_\sigma(r) = \frac{P_\sigma(r)}{\vpr(r)} \; ,
\qquad
P_\sigma(r) = \int_0^r dx \, \vpr(x) \, S_\sigma(x) \; ,
\label{eq.target}
\end{equation}
%
so only the integrated powers enter.  The sources depend on the
profiles, $S_\sigma = S_\sigma(T)$, through fusion heating,
electron-ion exchange and radiation.  The transport flux is
%
\begin{equation}
Q_\sigma = Q_\sigma^{\rm turb} + Q_\sigma^{\rm neo} \; ,
\end{equation}
%
with $Q^{\rm turb}_\sigma$ from TGLF and, at present,
$Q^{\rm neo}_\sigma = 0$.

The channels couple only pointwise, through the local flux and the
sources.  We therefore drop $\sigma$: $z$, $T$, $Q$ and $A$ are read as
vectors over the $n_c$ channels at each radius, and every operation
below acts on them componentwise.

\section{Normalization}

\begin{equation}
\qgb = n_e T_e c_s \left( \frac{\rho_s}{a} \right)^{2} \; ,
\qquad
c_s = \sqrt{\frac{T_e}{m_D}} \; ,
\qquad
\rho_s = \frac{c_s}{\Omega_{ci}} = \frac{m_D c_s c}{e B_{\rm unit}} \; ,
\end{equation}
%
with $a$ the minor radius of the last closed flux surface and $\Omega_{ci}
= e B_{\rm unit}/(m_D c)$ the deuterium cyclotron frequency (Gaussian
units).  Explicitly,
%
\begin{equation}
\qgb = \frac{m_D^{1/2} c^2}{e^2 a^2 B_{\rm unit}^2} \; n_e \, T_e^{5/2} \; .
\label{eq.qgb}
\end{equation}
%
Writing $\qhat \doteq Q/\qgb$, the normalized target is
%
\begin{equation}
\qhat^T = \frac{P}{\vpr \, \qgb} \; ,
\label{eq.qhattarget}
\end{equation}
%
which depends on the solution through $\qgb$ even when $S$ does not.

\section{Scale-length formulation}

With $x \doteq r/a$, the unknown is
%
\begin{equation}
z(x) \doteq - \frac{\partial \ln T}{\partial x} \; .
\label{eq.zdef}
\end{equation}
%
Integrating (\ref{eq.zdef}) outward from the origin,
%
\begin{equation}
\ln T(x) = \ln A - \int_0^{x} dx^\prime \, z(x^\prime) \; ,
\qquad
A \doteq T(0) \; ,
\label{eq.integrate}
\end{equation}
%
so $z$ determines the profile up to the single constant $A$.

\subsection{Axis condition}

$T$ is even in $x$, so $z$ is odd:
%
\begin{equation}
z(0) = 0 \; .
\label{eq.axis}
\end{equation}
%
This is imposed, not solved for.

\subsection{Grid and reintegration}

\begin{equation}
x_0 = 0 \; , \qquad x_1 < x_2 < \dots < x_N \; ,
\label{eq.grid}
\end{equation}
%
non-uniform, with resolution concentrated toward the edge.  The
integral in (\ref{eq.integrate}) is a linear functional of $z$, so for
any quadrature with weights $W$,
%
\begin{equation}
\boxed{\;
T_j(z,A) = A \exp \left( - \sum_{m=0}^{j} W_{jm} \, z_m \right) \; ,
\qquad z_0 = 0 \; . }
\label{eq.W}
\end{equation}
%
$W_{jm}$ is the cumulative quadrature weight for $\int_0^{x_j}$, so
$W_{jm} = 0$ for $m > j$: $W$ is lower triangular, and $T_j$ depends on
$z$ only at radii interior to $x_j$.  Positivity of $T$ is automatic.

\section{Boundary condition}
\label{sec.bc}

Any {\it linear} condition on the discrete profile $T = \{ T_j \}$ may
be written, without loss of generality, as
%
\begin{equation}
B \cdot T = 1 \; ,
\label{eq.linbc}
\end{equation}
%
with $B$ a known vector.  By (\ref{eq.W}), $T$ is linear in $A$, so
(\ref{eq.linbc}) determines $A$:
%
\begin{equation}
\boxed{\;
A = \left[ \sum_{j} B_j \, \exp \left(
 - \sum_{m=0}^{j} W_{jm} \, z_m \right) \right]^{-1} \; }
\label{eq.profile}
\end{equation}

\subsection{The classic case}

Write $T_*$ for the prescribed boundary value at $x_N$.  The Dirichlet
condition $T_N = T_*$ is $B_j = \delta_{jN}/T_*$.  Only the $j=N$ term
survives in (\ref{eq.profile}),
%
\begin{equation}
A = T_* \exp \left( \sum_{m=0}^{N} W_{Nm} \, z_m \right) \; ,
\end{equation}
%
and substituting into (\ref{eq.W}),
%
\begin{equation}
T_j = T_* \exp \left( \sum_{m=0}^{N}
 \bigl( W_{Nm} - W_{jm} \bigr) \, z_m \right) \; .
\label{eq.dirichlet}
\end{equation}
%
For the trapezoidal rule the cumulative weights are
%
\begin{equation}
W_{j0} = \frac{x_1-x_0}{2} \; , \qquad
W_{jm} = \frac{x_{m+1}-x_{m-1}}{2} \quad (0 < m < j) \; , \qquad
W_{jj} = \frac{x_j-x_{j-1}}{2} \; ,
\label{eq.trap}
\end{equation}
%
with $W_{jm} = 0$ for $m > j$.  The difference $W_{Nm} - W_{jm}$
collapses to the panels outside $x_j$, and (\ref{eq.dirichlet}) is
%
\begin{equation}
\boxed{\;
T_j = T_* \exp \left( \sum_{m=j+1}^{N}
 \frac{x_m - x_{m-1}}{2} \bigl( z_m + z_{m-1} \bigr) \right) \; }
\label{eq.dirichlettrap}
\end{equation}
%
the trapezoidal quadrature of $\int_{x_j}^{x_N} dx \, z$.  $T_j$ is
independent of $z$ interior to $x_j$, and (\ref{eq.axis}) never enters.

\section{Discrete problem}

For one channel the unknown is
%
\begin{equation}
z = \{ z_j \} \in \mathbb{R}^{N} \; ,
\qquad
z_j \doteq z(x_j) \; ,
\end{equation}
%
and the $N$ equations to be solved are
%
\begin{equation}
\boxed{\;
\qhat_j(z) = \qhat^T_j(z) \; }
\label{eq.match}
\end{equation}
%
both sides evaluated on the profile (\ref{eq.profile}) built from the
same $z$.  Over all channels the system has dimension $N n_c$.

\section{Structure}
\label{sec.structure}

The flux is local in the pair $(z_j,T_j)$,
%
\begin{equation}
\qhat_j = \qhat \bigl( z_j ;\, T_j \bigr) \; ,
\label{eq.localflux}
\end{equation}
%
with density, geometry and field fixed.  $T_j$ enters through $\tau =
T_i/T_e$, $\beta_e \propto n_e T_e/B_{\rm unit}^2$, $\nu_e \propto n_e
T_e^{-2}$, the Debye length, and $\qgb \propto n_e T_e^{5/2}$: weak, but
nonzero.

The locality is in $(z_j,T_j)$, not in $z$.  By (\ref{eq.W}) and
(\ref{eq.profile}), $T_j$ depends on $z$ interior to $x_j$ through $W$,
and on all of $z$ through $A$.  Hence $\qhat_j$ is a nonlinear, nonlocal
functional of $z$, and (\ref{eq.match}) is coupled across the grid.

The target is explicitly nonlocal,
%
\begin{equation}
\qhat^T_j = \frac{1}{\vpr_j \, \qgbj}
 \int_0^{x_j} dx \, \vpr(x) \, S(x) \; ,
\label{eq.nonlocaltarget}
\end{equation}
%
a functional of $T$ at every radius interior to $x_j$, since $S = S(T)$.

\section{A learned inverse}
\label{sec.inverse}

Existing surrogate methods (PORTALS) fit the {\it forward} map
(\ref{eq.localflux}), $z \mapsto \qhat$, and solve
%
\begin{equation}
\qhat^{\rm model}_j (z) = \qhat^T_j
\end{equation}
%
for $z$ by root-finding at every iteration.  Consider fitting the {\it
inverse} map instead.  At fixed $T_j$, (\ref{eq.localflux}) is $n_c$
equations in the $n_c$ unknowns $z_j$, and may be inverted:
%
\begin{equation}
\boxed{\;
z_j = \zeta \bigl( \qhat_j,\, T_j \bigr) \; }
\label{eq.inverse}
\end{equation}
%
$\zeta$ returns the gradient that carries a given flux.  Since $\qhat^T$
is known, the matching gradient is obtained by {\it evaluating} $\zeta$,
and (\ref{eq.match}) becomes a fixed point in the profile alone,
%
\begin{equation}
z^{(k+1)}_j = \zeta \bigl( \qhat^T_j(z^{(k)}),\, T_j(z^{(k)}) \bigr)
 \; ,
\label{eq.picard}
\end{equation}
%
with $T(z)$ from (\ref{eq.W}) and (\ref{eq.profile}).  Were $T$ exact,
one pass would suffice: the iteration resolves only the nonlocality of
$T_j(z)$, which (\ref{eq.inverse}) holds fixed, and whose effect on
$\qhat$ is weak.

The training data are those of the forward map: each flux evaluation at
$x_j$ yields $(z_j,T_j) \to \qhat_j$, read as input $(\qhat,T)$ and
output $z$.  $\zeta$ is a fixed function of its $2 n_c$ arguments, and
may be trained offline.

The flux is stiff: small below a critical gradient, rising steeply above
it.  $\qhat(z)$ is steep, so an error in a fitted $\qhat$ is amplified
on inversion; $\zeta(\qhat)$ is flat, and an error in $\zeta$ is not.
But $\zeta$ is single-valued only above the critical gradient: below it
$\qhat \simeq 0$ and the inverse is ill-posed.  The contraction rate of
(\ref{eq.picard}) is unmeasured.

\section{Fixed-source limit}
\label{sec.constsource}

Take a single ion channel ($n_c=1$), evolve only $z(x) = -\partial_x \ln
T_i$, and freeze the density, geometry, field and electron profile.  Let
the source be fixed in time: $S = S(x)$, a prescribed function of radius
that does not follow the profile.

\subsection{The target becomes fixed data}

The power is then a known quadrature, independent of the profile,
%
\begin{equation}
P(x) = \int_0^x dx^\prime \, \vpr(x^\prime) \, S(x^\prime) \; ,
\end{equation}
%
so from (\ref{eq.qhattarget})
%
\begin{equation}
\qhat^T_j = \frac{P(x_j)}{\vpr_j \, \qgbj} \;\equiv\; g_j \; .
\label{eq.gj}
\end{equation}
%
Since $\qgb \propto n_e T_e^{5/2}$ with $n_e,T_e$ fixed, $\qgbj$ is fixed,
and both $P(x_j)$ and $g_j$ are \emph{known constants, precomputed once}.
This is the payoff of a time-independent source: the target loses both
its profile dependence and the nonlinearity $S = S(T)$ that makes
(\ref{eq.nonlocaltarget}) nonlocal.

\subsection{The problem}

With the classic Dirichlet map (\ref{eq.dirichlettrap}) written as
%
\begin{equation}
T_j = T_* \exp\bigl( (Uz)_j \bigr) \; ,
\qquad
(Uz)_j = \sum_{m=j+1}^{N} \frac{x_m - x_{m-1}}{2} \, (z_m + z_{m-1}) \; ,
\label{eq.Umap}
\end{equation}
%
$U$ strictly upper triangular ($T_j$ depends on $z$ only \emph{exterior}
to $x_j$), the discrete problem (\ref{eq.match}) is
%
\begin{equation}
\boxed{\;
F_j(z) \doteq \qhat\bigl( z_j ;\, T_j(z) \bigr) - g_j = 0 \; ,
\qquad j = 1,\dots,N \; , }
\label{eq.constF}
\end{equation}
%
with $g$ fixed and the flux local in $(z_j,T_j)$ (\ref{eq.localflux}).

\subsection{Structure}

The Jacobian is diagonal plus a small one-sided coupling,
%
\begin{equation}
\frac{\partial F_j}{\partial z_k}
 = \partial_z \qhat \big|_j \, \delta_{jk}
 + \partial_T \qhat \big|_j \, U_{jk} \; ,
\end{equation}
%
the first term stiff and diagonal, the second weak and upper triangular.
Dropping the weak term decouples the system completely: with $\zeta$ the
local inverse (\ref{eq.inverse}),
%
\begin{equation}
z_j = \zeta(g_j) \qquad \text{independently at every node,}
\end{equation}
%
a one-shot solution with no coupling and no iteration.

The triangular $U$ admits an exact sweep from edge to axis.  At $j=N$,
$T_N = T_*$ exactly, so $z_N = \zeta(g_N, T_*)$.  Given $z_{j+1},\dots,z_N$,
the node value $T_j$ depends on the unknown $z_j$ through the last panel
alone,
%
\begin{equation}
T_j = T_* \exp\Bigl( c_j + \tfrac{1}{2} h_j \, z_j \Bigr) \; ,
\qquad
h_j = x_{j+1} - x_j \; ,
\end{equation}
%
with $c_j$ known, so each node is a \emph{scalar} fixed point
%
\begin{equation}
\boxed{\;
z_j = \zeta\Bigl( g_j,\; T_* \exp\bigl( c_j + \tfrac{1}{2} h_j z_j \bigr)
 \Bigr) \; , }
\label{eq.sweep}
\end{equation}
%
one scalar root-find per node in a fixed order.  The $N \times N$
nonlinear solve collapses to $N$ scalar problems coupled only through a
single quadrature panel.  Where $g_j$ falls below the critical gradient,
$\zeta$ is ill-posed ($\qhat \simeq 0$) and $z_j$ is pinned to the
critical gradient rather than solved.

\section{Edge boundary condition}

EGYRO solves to the separatrix.  ELM-free and without a pedestal, there
is no pedestal top at which to pin $T$, and $T_*$ is no longer
prescribed: it is set by parallel transport in the scrape-off layer,
%
\begin{equation}
T_* = f \bigl( P_{\rm sep} \bigr) \; ,
\qquad
T_* \sim P_{\rm sep}^{2/7} \; ,
\label{eq.sol}
\end{equation}
%
the exponent following from $q_\parallel = -\kappa_0 T^{5/2} \partial_s
T$.  But the power crossing the separatrix,
%
\begin{equation}
P_{\rm sep} = \int_0^1 dx \, \vpr(x) \, S(x) \; ,
\end{equation}
%
is a functional of the profile, since $S = S(T)$.  Hence $B$ in
(\ref{eq.linbc}) depends on $z$, and (\ref{eq.profile}) is unavailable:
$A$ can no longer be eliminated.  Promote it to an unknown and append
(\ref{eq.sol}),
%
\begin{align}
\qhat_j(z,A) = &~ \qhat^T_j(z,A) \; ,
 && N n_c \ \text{equations} \; , \\
T_N(z,A) - f \bigl( P_{\rm sep}[T(z,A)] \bigr)
 = &~ 0 \; , && n_c \ \text{equations} \; ,
\label{eq.augmented}
\end{align}
%
with $(z,A) \in \mathbb{R}^{N n_c + n_c}$.  The appended rows require no
flux evaluation, and the $2/7$ exponent makes the added nonlinearity
weak.

\section{Solution by active-learning surrogate}
\label{sec.surrogate}

Each evaluation of $\qhat$ is a gyrokinetic (or TGLF) run whose cost
exceeds everything else by orders of magnitude, so the count of flux
evaluations is the only cost that matters.  The sweep (\ref{eq.sweep})
is then better read not as $N$ independent root-finds but as recovery of
a single smooth curve $z^\star(x)$: with $g_j$ fixed data (\ref{eq.gj}),
the target is
%
\begin{equation}
\qhat\bigl( x,\, z^\star(x) \bigr) = g(x) \; ,
\label{eq.rootcurve}
\end{equation}
%
and the fluxes and gradients are smooth in radius.  The number of
evaluations should therefore scale with the complexity of $z^\star(x)$,
not with $N$.

\subsection{How the solver works}

Represent $z(x)$ through a cheap predictor $\widehat Q(x,z)$ of the flux
that also reports its uncertainty, and spend each slow, noisy call
$Q(x,z)$ where it teaches the most:
%
\begin{verbatim}
data = {}                     # runs so far: (x, z, Q)
seed with a few (x, z0, Q) at spread-out radii

repeat:
    fit Qhat(x,z) to data     # smooth in x; mean + error
    pick (x,z): near target Qhat=g, most uncertain
    data += (x, z, Q(x,z))    # one slow call
until confident everywhere

return z(x) where Qhat(x,z) = g(x)
\end{verbatim}
%
Smoothness in $x$ lets one call inform a band of radii, so a handful pin
the curve; the fit (not interpolation) averages out noise; and where $Q
\simeq 0$ the matching $z$ is undetermined and is pinned to the critical
gradient.

\subsection{Surrogate}

Model the forward flux as a Gaussian process over radius and gradient,
%
\begin{equation}
\boxed{\;
\qhat(x,z) \sim \mathcal{GP}\bigl( \mu,\;
 k_x(x,x^\prime)\, k_z(z,z^\prime) \bigr) \; , }
\label{eq.gp}
\end{equation}
%
the object the code actually returns: a gradient $z$ at radius $x$ is
mapped to one noisy $\qhat$.  The code is never inverted; the
\emph{surrogate} is, cheaply, in place of $\zeta$ (\ref{eq.inverse}).
The weak nonlocal dependence on $T_j(z)$ is lagged through the sweep
between surrogate updates, keeping (\ref{eq.gp}) two-dimensional; $T$ is
promoted to a third input only if the lag drifts.

\subsection{Priors}

Three physical facts enter as structure, and each lowers the evaluation
count:
%
\begin{itemize}
\item $k_x$ smooth (Mat\'ern-$5/2$ or squared-exponential): one run at
$x_j$ constrains the flux over a band of neighboring radii, decoupling
the evaluation count from $N$.
\item Monotonicity may be imposed only for a single channel: for one
species $\partial_z \qhat > 0$ above the critical gradient (stiffness),
which makes the surrogate root in (\ref{eq.rootcurve}) unique and
stabilizes the inversion.  For $n_c > 1$ the flux is a vector map
$\qhat_\sigma(z_1,\dots,z_{n_c})$ whose off-diagonal transport couples
the channels and can change sign; the Jacobian $\partial \qhat_\sigma /
\partial z_{\sigma^\prime}$ is not sign-definite, the map is not
monotone, and uniqueness is not guaranteed.  The multichannel surrogate
therefore rests on smoothness alone, with the local inversion resolving
branches from the current iterate.
\item The flat-then-steep threshold is nonstationary; fit
$\operatorname{asinh} \qhat$ or warp the $z$ input.  Below the critical
gradient $\qhat \simeq 0$ and the inverse is ill-posed
(\S\ref{sec.inverse}); the posterior variance diverges there and flags
it, and those radii are pinned to the critical gradient rather than
sampled.
\end{itemize}

\subsection{Acquisition}

The unknown is a contour, not an optimum: the zero level set of $\qhat -
g$.  Place each run where the posterior crossing is least certain, by the
straddle rule
%
\begin{equation}
\boxed{\;
(x,z)_{\rm next} = \arg\max \bigl[\, \sigma(x,z)
 - \kappa \, | \mu(x,z) - g(x) | \,\bigr] \; , }
\label{eq.straddle}
\end{equation}
%
with $\mu,\sigma$ the GP posterior mean and standard deviation.  Through
$k_x$ a single run collapses uncertainty across a neighborhood of radii.
Iterate until $\operatorname{Var}[ z^\star(x) ]$ is below tolerance
everywhere --- a principled minimum-evaluation stopping rule rather than
a fixed budget.

\subsection{Noise}

A GP with an observation-noise term regresses rather than interpolates,
which is required for the saturated-flux statistics of a gyrokinetic
run.  Take the noise heteroscedastic, $\sigma_n(x,z)$ growing toward
marginality where fluxes are small and intermittent; (\ref{eq.straddle})
then discounts those points automatically.  Noise tolerance is thus
intrinsic: it may be ignored initially and enabled later with no change
of structure.

\subsection{Relation to the forward and inverse maps}

This is the disciplined form of a PORTALS-style surrogate: the same
forward fit (\ref{eq.localflux}), but the acquisition targets the root
contour and the kernel carries radial smoothness (and, for a single
channel, monotonicity), so far fewer runs are needed.  It realizes the
inverse of \S\ref{sec.inverse} safely --- the forward surrogate is
inverted locally, only where (\ref{eq.straddle}) has already judged a run
worthwhile, avoiding the ill-posed branch of a globally fitted $\zeta$.  The triangular sweep
(\ref{eq.sweep}) is the cheap inner correction that threads the $T_j(z)$
coupling between surrogate updates.
